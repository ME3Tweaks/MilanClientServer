MISON - Milan Structured Object Notation
------------------------------------------

Small binary format designed for serializing contents of client / server
communication for Project Milan. Originally based on Mongo's BSON and EA's TDF.

  * All numerics must be serialized in big-endian byte order.
  * All textual values must be valid UTF-8, narrowed down to ASCII when specified.
  * While a document is entirely self-describing, individual elements are no longer so.
  * Syntax of this spec is not supposed to be valid BNF.


document    ::= names element           ; a complete message

names       ::= uint16_t('count) (name_entry * 'count)              ; element name table
name_entry  ::= byte_t('size + 1) (byte_t * 'size) byte_t(0x00)     ; an ascii element name string

element     ::= type uint16_t(0xffff) value     ; anonymous element
              | type uint16_t('index) value     ; named element, with 'index into name table

type        ::= byte_t(0x00)            ; None, an absence of value.
              | byte_t(0x01)            ; Object, a collection of elements.
              | byte_t(0x02)            ; Array, a collection of values.
              | byte_t(0x03)            ; Map, a collection of value -> value pairs.
              | byte_t(0x04)            ; Int32, a signed 32-bit integer value.
              | byte_t(0x05)            ; Int64, a signed 64-bit integer value.
              | byte_t(0x06)            ; Single, a single-precision 32-bit floating-point value.
              | byte_t(0x07)            ; Double, a double-precision 64-bit floating-point value.
              | byte_t(0x08)            ; Bool, an 8-bit boolean value.
              | byte_t(0x09)            ; String, a UTF-8 string.
              | byte_t(0x0A)            ; Binary, an untyped sequence of bytes.

value       ::= ()                      ; type is "None"
              | object_value_t          ; type is "Object"
              | array_value_t           ; type is "Array"
              | map_value_t             ; type is "Map"
              | int32_t                 ; type is "Int32"
              | int64_t                 ; type is "Int64"
              | float32_t               ; type is "Single"
              | float64_t               ; type is "Double"
              | byte_t(0x00)            ; type is "Bool", meaning "false"
              | byte_t(0x01)            ; type is "Bool", meaning "true"
              | string_value_t          ; type is "String"
              | blob_value_t            ; type is "Binary"

string_value_t  ::= uint32_t('size + 1) (byte_t * 'size) byte_t(0x00)
blob_value_t    ::= uint32_t('size) (byte_t * 'size)

   ; in all definitions below 'size means number of bytes until end of value

object_value_t  ::= uint32_t('size) uint16_t('count) (element * 'count)
array_value_t   ::= uint32_t('size) uint16_t('count) type('value_type) (value * 'count)

map_value_t     ::= uint32_t('size) uint16_t('count)
                    type('key_type) type('value_type)
                    ((value('key) value('value)) * 'count)
